Data Descriptions

All three datasets share the following filters:

  • HLA restriction: Only HLA-A*02:01 epitopes are included
  • Assay type: Eluted ligand data only

Additional filters per dataset:

Dataset Source
Viral Viral antigens only, human samples only, no mouse samples
Cancer Epitopes associated with occurrence of cancer
Healthy Epitopes from healthy individuals, no immunization

# Helper function to produce a length summary table
length_summary <- function(df, label) {
  df |>
    count(pep_length, name = "count") |>
    arrange(pep_length) |>
    mutate(
      percentage = round(count / sum(count) * 100, 2),
      dataset    = label
    )
}

bind_rows(
  length_summary(pos_viral_proteins, "Viral"),
  length_summary(pos_cancer,         "Cancer"),

) |>

    "Peptide Length" = pep_length,
    "Count"          = count,
    "Percentage (%)" = percentage
  ) |>

Difference Logo Plots — 9-mer Comparisons

Difference logo plots show enrichment (positive, above line) and depletion (negative, below line) of amino acids at each position relative to a reference. Here each dataset is compared pairwise against the others.

# Helper: compute position probability matrix from sequences
get_ppm <- function(seqs) {
  # All amino acids
  aa <- c("A","C","D","E","F","G","H","I","K","L",
          "M","N","P","Q","R","S","T","V","W","Y")
  
  seq_matrix <- do.call(rbind, strsplit(seqs, ""))
  ppm <- apply(seq_matrix, 2, function(col) {
    counts <- table(factor(col, levels = aa))
    counts / sum(counts)
  })
  rownames(ppm) <- aa
  ppm
}

# Helper: compute difference PPM and plot
plot_diff_logo <- function(seqs_a, seqs_b, label_a, label_b) {
  ppm_a    <- get_ppm(seqs_a)
  ppm_b    <- get_ppm(seqs_b)
  diff_ppm <- ppm_a - ppm_b

  ggseqlogo(diff_ppm, method = "custom") +
    labs(
      title    = paste0("Difference Logo — ", label_a, " vs. ", label_b),
      subtitle = paste0(
        label_a, " (n = ", length(seqs_a), ") minus ",
        label_b, " (n = ", length(seqs_b), ")"
      ),
      x = "Position",
      y = "Difference in Probability"
    ) +
    geom_hline(yintercept = 0, linetype = "dashed", colour = "grey40") +
    theme_logo() +
    theme(
      plot.title    = element_text(hjust = 0.5, size = 14, face = "bold"),
      plot.subtitle = element_text(hjust = 0.5, size = 11)
    )
}
